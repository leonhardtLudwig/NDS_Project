\subsection{Introduction to the FDG model}
\label{subsec:fdg-intro}

\subsubsection{The FDG model}
\label{subsubsec:fdg-model}

Consider a group of individuals $V = \{1, 2, \dots, n\}$, where $x_i \in \mathbb{R}$
is the opinion that individual $i$ holds on a given topic. The influences among the
individuals are described by a weighted digraph
$G = (V, E, \{a_{ij}\}_{(i,j) \in E})$: an edge $(i,j) \in E$, with weight
$a_{ij} > 0$, means that individual $i$ is influenced by individual $j$.

The French--Harary--DeGroot (FDG) model describes how each individual, or agent,
updates its opinion: the new opinion is a weighted average of the opinions of the
agents it is influenced by,
\begin{equation}\label{eq:fdg}
  x_i(k+1) = \sum_{j=1}^{n} a_{ij}\, x_j(k), \qquad \sum_{j=1}^{n} a_{ij} = 1,
\end{equation}
or, in vector form, $x(k+1) = A x(k)$, where $A$ is the weighted adjacency matrix
of $G$ (Definition~\ref{def:adjacency}). The constraint on the weights means that
$A$ is row-stochastic, i.e.\ $A$ is non-negative and $A \mathbf{1}_n = \mathbf{1}_n$.

\begin{theorem}[Spectral properties of row-stochastic matrices]
\label{thm:stoch-spectrum}
If $A \in \mathbb{R}^{n \times n}$ is row-stochastic, then
$1 \in \operatorname{spec}(A)$ and $\rho(A) = 1$; hence $\operatorname{spec}(A)$ is
contained in the closed unit disk.
\end{theorem}

\begin{definition}[Semi-convergent matrix]
A matrix $A \in \mathbb{R}^{n \times n}$ is \emph{semi-convergent} if
$\lim_{k \to +\infty} A^k$ exists and is finite.

\end{definition}

Since $x(k) = A^k x(0)$, the asymptotic behaviour of the model is determined by the
powers of $A$. Three outcomes are possible:
\begin{itemize}
  \item \emph{no convergence}: $\lim_{k \to +\infty} x(k)$ does not exist for some
        $x(0)$; this happens when $A$ is not semi-convergent, e.g.\ when $A$ has an
        eigenvalue $\mu \neq 1$ with $|\mu| = 1$;
  \item \emph{consensus}: $\lim_{k \to +\infty} x(k) = \alpha \mathbf{1}_n$ for some
        $\alpha \in \mathbb{R}$;
  \item \emph{disagreement}: $\lim_{k \to +\infty} x(k) = v$ with
        $v \neq \alpha \mathbf{1}_n$ for every $\alpha \in \mathbb{R}$.
\end{itemize}


\subsubsection{Graph-theoretic tools}
\label{subsubsec:graph-tools}

The outcome of the model depends on the spectrum of $A$, which in turn depends on the
structure of $G$. We recall the graph notions needed to link the two.

\begin{definition}[Weighted adjacency matrix]
\label{def:adjacency}
Given a weighted digraph $G = (V, E, \{a_{ij}\}_{(i,j) \in E})$ of order $n$, its
\emph{weighted adjacency matrix} is the $n \times n$ matrix $A$ whose $(i,j)$ entry
is equal to the edge weight $a_{ij}$ if $(i,j) \in E$, and is equal to zero
otherwise.
\end{definition}

\begin{definition}[Strong connectivity and globally reachable nodes]
A digraph $G = (V, E)$ is \emph{strongly connected} if there is a directed walk from
any node to any other node. A node of $G$ is \emph{globally reachable} if it can be
reached from any other node through a directed walk.
\end{definition}

With our convention, a directed walk from $i$ to $j$ means that agent $i$ is
influenced by agent $j$, either directly or through intermediaries. Hence $G$ is
strongly connected when every agent influences every other agent, at least
indirectly, and a globally reachable node is an agent that influences the whole
group.

\begin{definition}[Periodic digraph]
A strongly connected digraph is \emph{periodic} if there exists an integer $k > 1$
(the period) that divides the length of every simple cycle of the digraph; otherwise
it is \emph{aperiodic}.
\end{definition}

Equivalently, a strongly connected digraph $G$ is periodic if and only if the greatest
common divisor of the lengths of its simple cycles is larger than $1$. In particular,
a strongly connected digraph with a self-loop is aperiodic, since the self-loop is a
cycle of length $1$.

These notions characterize the matrix properties that determine the spectrum of $A$.

\begin{definition}[Irreducible and primitive matrices]
A non-negative matrix $A \in \mathbb{R}^{n \times n}$, $n \geq 2$, is
\emph{irreducible} if $\sum_{k=0}^{n-1} A^k$ is positive, and \emph{primitive} if
there exists $k \in \mathbb{N}$ such that $A^k$ is positive.
\end{definition}

\begin{theorem}[Irreducible matrices and digraphs]
\label{thm:irreducible}
Let $G$ be a weighted digraph of order $n \geq 2$ with weighted adjacency matrix $A$.
Then $A$ is irreducible if and only if $G$ is strongly connected.
\end{theorem}

\begin{theorem}[Primitive matrices and digraphs]
\label{thm:primitive}
Let $G$ be a weighted digraph of order $n \geq 2$ with weighted adjacency matrix $A$.
Then $A$ is primitive if and only if $G$ is strongly connected and aperiodic. In
particular, if $G$ is strongly connected and has at least one self-loop, then $A$ is
primitive.
\end{theorem}

\begin{definition}[Strongly connected components and condensation digraph]
Given a weighted digraph $G$, a subgraph $S$ is a \emph{strongly connected component}
of $G$ if $S$ is strongly connected and any other subgraph of $G$ strictly containing
$S$ is not strongly connected. The \emph{condensation digraph} $C(G)$ is obtained by
collapsing each strongly connected component of $G$ into a single node: for any two
distinct strongly connected components $S_k$ and $S_h$, there is an edge
$(S_k, S_h)$ in $C(G)$ if and only if there exists an edge $(i,j) \in E$ with
$i \in S_k$ and $j \in S_h$ (self-loops are not allowed in $C(G)$). A node of $C(G)$
with no outgoing edges is called a \emph{sink}.
\end{definition}

A sink of $C(G)$ is a group of agents that are influenced only by members of the group
itself: no agent outside the group can change their opinions.

If $G$ contains a globally reachable node, then $C(G)$ has a unique sink.


\subsubsection{Consensus and social power}
\label{subsubsec:consensus}

\begin{theorem}[Powers of row-stochastic matrices with a simple and strictly dominant eigenvalue]
\label{thm:stoch-powers}
Let $A$ be row-stochastic. If the eigenvalue $1$ is simple and strictly larger in
modulus than all the other eigenvalues of $A$, then $A$ is semi-convergent and
\[
  \lim_{k \to +\infty} A^k = \mathbf{1}_n w^\top ,
\]
where $w \geq 0$ is the dominant left eigenvector of $A$, normalized so that
$w^\top \mathbf{1}_n = 1$. The limit $\mathbf{1}_n w^\top$ is a rank-one projection
matrix.
\end{theorem}
